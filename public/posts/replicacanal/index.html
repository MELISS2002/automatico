import re

import srt

from datetime import timedelta



def capitalize_first_letter(word):

    if not word:

        return word

    return word[0].upper() + word[1:].lower()



def clean_word(word):

    cleaned = re.sub(r'[.,!?;:Â¿Â¡"]', '', word)

    return capitalize_first_letter(cleaned)



def split_srt_to_words(input_file, output_file):

    print(f"Procesando archivo: {input_file}")



    try:

        with open(input_file, "r", encoding="utf-8") as file:

            content = file.read()

            print("Contenido del archivo leÃ­do correctamente")

            subtitles = list(srt.parse(content))

    except Exception as e:

        print(f"Error al leer el archivo: {str(e)}")

        return



    if not subtitles:

        print("El archivo no contiene subtÃ­tulos o estÃ¡ vacÃ­o")

        return



    print(f"Encontrados {len(subtitles)} subtÃ­tulos para procesar")



    new_subtitles = []

    index = 1



    for i, subtitle in enumerate(subtitles, 1):

        print(f"\nProcesando subtÃ­tulo {i}:")

        print(f"Texto original: '{subtitle.content}'")

        print(f"DuraciÃ³n original: {subtitle.end - subtitle.start}")



        words = re.findall(r"\w+[\w'-]*", subtitle.content)

        cleaned_words = [clean_word(w) for w in words if clean_word(w)]

        print(f"Palabras procesadas: {cleaned_words}")



        if not cleaned_words:

            print("No se encontraron palabras vÃ¡lidas en este subtÃ­tulo")

            continue



        total_chars = sum(len(w) for w in cleaned_words)

        if total_chars == 0:

            continue



        duration_per_char = (subtitle.end - subtitle.start) / total_chars

        start_time = subtitle.start



        for word in cleaned_words:

            word_duration = duration_per_char * len(word)

            end_time = start_time + word_duration



            if i < len(subtitles) and end_time > subtitles[i].start:

                end_time = subtitles[i].start - timedelta(milliseconds=1)



            new_sub = srt.Subtitle(index, start_time, end_time, word)

            new_subtitles.append(new_sub)

            print(f"AÃ±adida palabra {index}: '{word}' ({start_time} --> {end_time})")



            start_time = end_time

            index += 1



    if not new_subtitles:

        print("No se generaron subtÃ­tulos. Verifica el formato del archivo.")

        return



    print(f"\nTotal de palabras generadas: {len(new_subtitles)}")



    try:

        with open(output_file, "w", encoding="utf-8") as file:

            file.write(srt.compose(new_subtitles))

        print(f"\nArchivo de salida '{output_file}' generado con Ã©xito!")

    except Exception as e:

        print(f"Error al guardar el archivo: {str(e)}")
